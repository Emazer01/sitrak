CREATE TABLE IF NOT EXISTS sesi (
    id_sesi SERIAL NOT NULL,
    nama_sesi VARCHAR(150) NOT NULL,
    nama_tempat_menembak VARCHAR(150) NOT NULL,
    jumlah_jalur INT NOT NULL DEFAULT 1,
    total_gelombang INT NOT NULL DEFAULT 1,
    jarak_meter INT NOT NULL DEFAULT 25,
    id_senjata INT NOT NULL,
    id_mode_tembakan INT NOT NULL,
    id_sikap_menembak INT NOT NULL,
    id_tipe_target INT NOT NULL,
    tanggal DATE NOT NULL DEFAULT CURRENT_DATE,
    status status_sesi_enum DEFAULT 'Menunggu',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY(id_sesi),
    FOREIGN KEY(id_senjata) REFERENCES senjata(id_senjata),
    FOREIGN KEY(id_mode_tembakan) REFERENCES mode_tembakan(id_mode_tembakan),
    FOREIGN KEY(id_sikap_menembak) REFERENCES sikap_menembak(id_sikap_menembak),
    FOREIGN KEY(id_tipe_target) REFERENCES tipe_target(id_tipe_target)
);
