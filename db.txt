-- ---------------------------------------------------------------------
-- 0. Trigger Function untuk Auto-Update updated_at
-- ---------------------------------------------------------------------
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ---------------------------------------------------------------------
-- 1. Custom Enum Types
-- ---------------------------------------------------------------------
DO $$ BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'user_role_enum') THEN
        CREATE TYPE user_role_enum AS ENUM ('admin', 'instruktur', 'operator', 'penembak');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_aktif_enum') THEN
        CREATE TYPE status_aktif_enum AS ENUM ('Aktif', 'Nonaktif');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'jenis_kelamin_enum') THEN
        CREATE TYPE jenis_kelamin_enum AS ENUM ('L', 'P');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'kondisi_senjata_enum') THEN
        CREATE TYPE kondisi_senjata_enum AS ENUM ('baik', 'perawatan', 'rusak');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'tipe_target_enum') THEN
        CREATE TYPE tipe_target_enum AS ENUM (
            'Lingkaran Konsentris', 
            'Siluet L1', 
            'Siluet Setengah Badan', 
            'Plat Baja / Popper', 
            'Custom'
        );
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'sikap_menembak_enum') THEN
        CREATE TYPE sikap_menembak_enum AS ENUM ('Berdiri', 'Berlutut', 'Tiarap', 'Kombinasi');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'mode_tembakan_enum') THEN
        CREATE TYPE mode_tembakan_enum AS ENUM ('Presisi', 'Reaksi', 'Double Tap', 'Penilaian');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_sesi_enum') THEN
        CREATE TYPE status_sesi_enum AS ENUM ('menunggu', 'berlangsung', 'selesai', 'dibatalkan');
    END IF;
    IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'kategori_kelulusan_enum') THEN
        CREATE TYPE kategori_kelulusan_enum AS ENUM ('Istimewa', 'Mahir', 'Cukup', 'Perlu Pembinaan');
    END IF;
END $$;

-- ---------------------------------------------------------------------
-- 2. Tabel: users (Autentikasi & Akun Pengguna)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS users (
    id_user SERIAL PRIMARY KEY AUTO INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    nama_lengkap VARCHAR(100) NOT NULL,
    role user_role_enum NOT NULL DEFAULT 'operator',
    status status_aktif_enum NOT NULL DEFAULT 'aktif',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE OR REPLACE TRIGGER trg_users_updated_at
BEFORE UPDATE ON users
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- ---------------------------------------------------------------------
-- 3. Tabel: personel (Data Anggota / Personel Penembak)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS personel (
    id_personel SERIAL PRIMARY KEY AUTO INCREMENT,
    nrp VARCHAR(30) NOT NULL UNIQUE,
    nama_lengkap VARCHAR(100) NOT NULL,
    pangkat VARCHAR(50) NOT NULL,
    satuan VARCHAR(100) NOT NULL,
    status status_aktif_enum DEFAULT 'aktif',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_personel_nrp ON personel(nrp);
CREATE INDEX IF NOT EXISTS idx_personel_satuan ON personel(satuan);

CREATE OR REPLACE TRIGGER trg_personel_updated_at
BEFORE UPDATE ON personel
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- ---------------------------------------------------------------------
-- 4. Tabel: senjata (Master Data Senjata Latihan)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS senjata (
    id_senjata SERIAL PRIMARY KEY AUTO INCREMENT,
    kode_senjata VARCHAR(50) NOT NULL UNIQUE,
    model_senjata VARCHAR(100) NOT NULL,
    jenis_senjata VARCHAR(50) NOT NULL CHECK (jenis_senjata IN ('Senapan Serbu', 'Pistol', 'Submachine Gun', 'Sniper / SPR', 'Lainnya')),
    kaliber VARCHAR(50) NOT NULL,
    pabrikan VARCHAR(100) DEFAULT 'PT Pindad',
    kapasitas_magazen INT DEFAULT 30,
    kondisi kondisi_senjata_enum DEFAULT 'baik',
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE OR REPLACE TRIGGER trg_senjata_updated_at
BEFORE UPDATE ON senjata
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- ---------------------------------------------------------------------
-- 5. Tabel: sasaran_target (Master Data Jenis Target / Sasaran Tembak)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS sasaran_target (
    id_target SERIAL PRIMARY KEY,
    nama_target VARCHAR(100) NOT NULL,
    tipe_target tipe_target_enum NOT NULL,
    diameter_cm NUMERIC(6,2) NULL,
    jumlah_ring INT DEFAULT 10,
    skor_maksimal INT DEFAULT 100,
    keterangan TEXT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------------------
-- 6. Tabel: sesi_latihan (Informasi Sesi Latihan Menembak)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS sesi_latihan (
    id_sesi SERIAL PRIMARY KEY,
    kode_sesi VARCHAR(50) NOT NULL UNIQUE,
    nama_sesi VARCHAR(150) NOT NULL,
    id_personel INT NOT NULL REFERENCES personel(id_personel) ON UPDATE CASCADE ON DELETE RESTRICT,
    id_senjata INT NOT NULL REFERENCES senjata(id_senjata) ON UPDATE CASCADE ON DELETE RESTRICT,
    id_target INT NOT NULL REFERENCES sasaran_target(id_target) ON UPDATE CASCADE ON DELETE RESTRICT,
    lane_jalur VARCHAR(20) NOT NULL,
    jarak_meter INT NOT NULL DEFAULT 25,
    sikap_menembak sikap_menembak_enum DEFAULT 'Berdiri',
    mode_tembakan mode_tembakan_enum DEFAULT 'Presisi',
    jumlah_peluru INT NOT NULL DEFAULT 10,
    timer_detik INT NULL,
    catatan TEXT NULL,
    status status_sesi_enum DEFAULT 'menunggu',
    waktu_mulai TIMESTAMPTZ NULL,
    waktu_selesai TIMESTAMPTZ NULL,
    total_skor NUMERIC(6,2) DEFAULT 0.00,
    akurasi_persen NUMERIC(5,2) DEFAULT 0.00,
    created_by INT NULL REFERENCES users(id_user) ON UPDATE CASCADE ON DELETE SET NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_sesi_kode ON sesi_latihan(kode_sesi);
CREATE INDEX IF NOT EXISTS idx_sesi_status ON sesi_latihan(status);

CREATE OR REPLACE TRIGGER trg_sesi_updated_at
BEFORE UPDATE ON sesi_latihan
FOR EACH ROW EXECUTE FUNCTION set_updated_at();

-- ---------------------------------------------------------------------
-- 7. Tabel: detail_tembakan (Tracking Tiap Butir Tembakan / Shot-by-Shot)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS detail_tembakan (
    id_tembakan BIGSERIAL PRIMARY KEY,
    id_sesi INT NOT NULL REFERENCES sesi_latihan(id_sesi) ON DELETE CASCADE ON UPDATE CASCADE,
    shot_number INT NOT NULL,
    koordinat_x NUMERIC(8,3) NOT NULL,
    koordinat_y NUMERIC(8,3) NOT NULL,
    jarak_dari_pusat NUMERIC(8,3) NULL,
    skor_poin NUMERIC(4,1) NOT NULL DEFAULT 0.0,
    ring_zona VARCHAR(20) NULL,
    is_hit BOOLEAN DEFAULT TRUE,
    split_time_detik NUMERIC(6,3) NULL,
    elapsed_time_detik NUMERIC(8,3) NULL,
    waktu_tembak TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

COMMENT ON COLUMN detail_tembakan.koordinat_x IS 'Posisi horizontal target (mm / px)';
COMMENT ON COLUMN detail_tembakan.koordinat_y IS 'Posisi vertikal target (mm / px)';
COMMENT ON COLUMN detail_tembakan.jarak_dari_pusat IS 'Jarak Euclidean dari pusat bullseye';
COMMENT ON COLUMN detail_tembakan.skor_poin IS 'Nilai poin tembakan: 10, 9, 8, dst.';
COMMENT ON COLUMN detail_tembakan.ring_zona IS 'Zona cincin atau zona alfabet A, C, D';
COMMENT ON COLUMN detail_tembakan.split_time_detik IS 'Interval waktu dari tembakan sebelumnya';
COMMENT ON COLUMN detail_tembakan.elapsed_time_detik IS 'Total waktu berlalu sejak tembakan pertama';

CREATE INDEX IF NOT EXISTS idx_detail_tembakan_sesi_shot ON detail_tembakan(id_sesi, shot_number);

-- ---------------------------------------------------------------------
-- 8. Tabel: evaluasi_kemampuan (Penilaian Radar Parameter Personel)
-- ---------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS evaluasi_kemampuan (
    id_evaluasi SERIAL PRIMARY KEY,
    id_personel INT NOT NULL REFERENCES personel(id_personel) ON DELETE CASCADE ON UPDATE CASCADE,
    id_sesi INT NULL REFERENCES sesi_latihan(id_sesi) ON DELETE SET NULL ON UPDATE CASCADE,
    skor_akurasi_bidik NUMERIC(5,2) DEFAULT 0.00,
    skor_kecepatan_reaksi NUMERIC(5,2) DEFAULT 0.00,
    skor_stabilitas_sikap NUMERIC(5,2) DEFAULT 0.00,
    skor_jarak_jauh NUMERIC(5,2) DEFAULT 0.00,
    skor_kendali_recoil NUMERIC(5,2) DEFAULT 0.00,
    skor_konsistensi NUMERIC(5,2) DEFAULT 0.00,
    nilai_akhir NUMERIC(5,2) DEFAULT 0.00,
    kategori_kelulusan kategori_kelulusan_enum DEFAULT 'Cukup',
    catatan_instruktur TEXT NULL,
    tanggal_evaluasi DATE NOT NULL DEFAULT CURRENT_DATE
);
